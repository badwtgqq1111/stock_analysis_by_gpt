    :return: 股票代码列表
    """

        current_date = datetime.datetime.now().date()
        all_stocks = get_all_securities(types='stock', date=current_date)
        # 过滤掉退市股票
        stock_list = all_stocks[all_stocks['end_date'] > current_date].index.tolist()

def query_top10_floating_shareholder_single(code, n=30):
    """
    查询单只股票的十大流通股东数据
    :param code: 股票代码
    :param n: 获取的记录条数，默认为30
    :return: 包含该股票数据的DataFrame
    """
    try:
        # 构建查询语句
        q = query(finance.STK_SHAREHOLDER_FLOATING_TOP10).filter(
            finance.STK_SHAREHOLDER_FLOATING_TOP10.code == code
        ).order_by(
            finance.STK_SHAREHOLDER_FLOATING_TOP10.pub_date.desc()
        ).limit(n)
        
        df = finance.run_query(q)
        return df
    except Exception as e:
        print(f"查询股票 {code} 时出错: {e}")
        return pd.DataFrame()

def save_to_excel_final(data_df, file_index):
    """
    修复版保存数据到Excel
    :param data_df: 要保存的数据DataFrame
    :param file_index: 文件序号
    """
    if data_df is None or data_df.empty:
        print("数据为空，跳过保存。")
        return False
    
    # 定义中文字段名映射
    column_mapping = {
        'company_id': '公司ID',
        'company_name': '公司名称',
        'code': '股票代码',
        'end_date': '截止日期',
        'pub_date': '公告日期',
        'change_reason_id': '变动原因编码',
        'change_reason': '变动原因',
        'shareholder_rank': '股东名次',
        'shareholder_id': '股东ID',
        'shareholder_name': '股东名称',
        'shareholder_name_en': '股东名称(英文)',
        'shareholder_class_id': '股东类别编码',
        'shareholder_class': '股东类别',
        'share_number': '持股数量',
        'share_ratio': '持股比例',
        'sharesnature_id': '股份性质编码',
        'sharesnature': '股份性质'
    }
    
    # 重命名列
    data_df.rename(columns=column_mapping, inplace=True)
    
    # 定义保存路径和文件名
    save_dir = 'data'
    if not os.path.exists(save_dir):
        os.makedirs(save_dir)
    
    # 修复点1：统一文件名格式，删除"自定义"字样
    file_name = f'十大流通股东-{file_index:04d}.xlsx'
    file_path = os.path.join(save_dir, file_name)
    
    try:
        # 创建新的工作簿
        wb = Workbook()
        ws = wb.active
        ws.title = '十大流通股东'
        
        # 写入表头
        headers = list(data_df.columns)
        for col_idx, header in enumerate(headers, 1):
            ws.cell(row=1, column=col_idx, value=header)
        
        # 写入数据
        for row_idx, row_data in enumerate(data_df.values, 2):
            for col_idx, cell_value in enumerate(row_data, 1):
                ws.cell(row=row_idx, column=col_idx, value=cell_value)
        
        # 设置全局字体和自动换行
        font = Font(name='宋体', size=10)
        alignment = Alignment(wrap_text=True, vertical='center')
        
        for row in ws.iter_rows(min_row=1, max_row=ws.max_row, min_col=1, max_col=ws.max_column):
            for cell in row:
                cell.font = font
                cell.alignment = alignment
        
        # 设置行高
        for row in range(1, ws.max_row + 1):
            ws.row_dimensions[row].height = 25
        
        # 设置列宽
        column_width_config = [
            ('公司ID', 9),
            ('公司名称', 16),
            ('股票代码', 10),
            ('截止日期', 12),
            ('公告日期', 12),
            ('变动原因编码', 9),
            ('变动原因', 10),
            ('股东名次', 4),
            ('股东ID', 9),
            ('股东名称', 16),
            ('股东名称(英文)', 16),
            ('股东类别编码', 9),
            ('股东类别', 16),
            ('持股数量', 14),
            ('持股比例', 9),
            ('股份性质编码', 9),
            ('股份性质', 10)
        ]
        
        for col_idx, (col_name, width) in enumerate(column_width_config, 1):
            col_letter = get_column_letter(col_idx)
            ws.column_dimensions[col_letter].width = width
        
        # 设置日期格式
        date_format = 'YYYY-MM-DD'
        
        # 修复点2：直接设置单元格的数字格式，而不是通过NamedStyle
        # 找到截止日期和公告日期列的索引
        date_columns = []
        for idx, col_name in enumerate(column_width_config, 1):
            if col_name[0] in ['截止日期', '公告日期']:
                # 查找这个列名在DataFrame中的实际位置
                for data_col_idx, data_col_name in enumerate(headers, 1):
                    if data_col_name == col_name[0]:
                        date_columns.append(data_col_idx)
                        break
        
        # 为日期列设置格式
        for date_col_idx in date_columns:
            col_letter = get_column_letter(date_col_idx)
            for cell in ws[col_letter][1:]:  # 跳过标题行
                cell.number_format = date_format
        
        # 保存工作簿
        wb.save(file_path)
        
        print(f"✓ 数据已保存至：{file_path}，共 {len(data_df)} 行。")
        return True
    except Exception as e:
        print(f"保存文件时出错: {e}")
        return False

def process_stocks_final(stock_list):
    """
    最终版本的处理股票列表函数
    :param stock_list: 股票列表
    """
    if not stock_list:
        print("股票列表为空，跳过处理。")
        return
    
    # 性能优化参数
    batch_size = 300  # 每批次处理300只股票
    stocks_per_chunk = 30  # 每处理30只股票清理一次内存
    batches_per_save = 3  # 每3批保存一次
    file_index = 1  # 文件序号
    
    # 存储每批数据的列表
    batch_dataframes = []
    
    total_processed_stocks = 0
    total_data_rows = 0
    no_data_stocks = []
    error_stocks = []
    
    # 批次计数器
    batch_counter = 0
    
    print(f"最终优化处理，共 {len(stock_list)} 只股票")
    print(f"批次大小: {batch_size} 只/批，每 {batches_per_save} 批保存一次")
    print(f"内存清理频率: 每 {stocks_per_chunk} 只股票清理一次")
    
    for i in range(0, len(stock_list), batch_size):
        batch_counter += 1
        batch = stock_list[i:i+batch_size]
        current_batch = i//batch_size + 1
        
        print(f"\n[批次 {current_batch}] 处理 {len(batch)} 只股票...")
        
        # 创建本批次的DataFrame列表，分块处理
        batch_chunks = []
        chunk_data = []
        chunk_counter = 0
        
        batch_no_data = []
        
        for idx, code in enumerate(batch, 1):
            try:
                # 每处理stocks_per_chunk只股票显示一次进度
                if idx % stocks_per_chunk == 0 or idx == len(batch):
                    print(f"  本批进度: {idx}/{len(batch)}")
                    # 清理内存
                    gc.collect()
                
                # 查询数据
                df = query_top10_floating_shareholder_single(code, n=30)
                
                if not df.empty:
                    chunk_data.append(df)
                    chunk_counter += 1
                    
                    # 每stocks_per_chunk只股票合并一次，减少DataFrame数量
                    if chunk_counter >= stocks_per_chunk:
                        if chunk_data:
                            chunk_df = pd.concat(chunk_data, ignore_index=True)
                            batch_chunks.append(chunk_df)
                            chunk_data = []
                            chunk_counter = 0
                            # 清理内存
                            gc.collect()
                    
                    if len(df) < 10:  # 如果数据很少，也记录一下
                        print(f"  {code}: 获取到 {len(df)} 条记录")
                else:
                    batch_no_data.append(code)
                    no_data_stocks.append(code)
                
                # 删除临时变量
                del df
                
            except Exception as e:
                print(f"  处理股票 {code} 时出错: {e}")
                error_stocks.append(code)
                continue
        
        # 处理剩余的chunk数据
        if chunk_data:
            chunk_df = pd.concat(chunk_data, ignore_index=True)
            batch_chunks.append(chunk_df)
            chunk_data = []
            gc.collect()
        
        # 记录本批无数据的股票
        if batch_no_data:
            print(f"  本批有 {len(batch_no_data)} 只股票无数据")
        
        total_processed_stocks += len(batch)
        
        if batch_chunks:
            # 合并本批次的所有chunk
            batch_df = pd.concat(batch_chunks, ignore_index=True)
            batch_data_rows = len(batch_df)
            total_data_rows += batch_data_rows
            print(f"  本批次获取 {batch_data_rows} 行数据，累计 {total_data_rows} 行")
            
            # 将本批次DataFrame添加到列表中
            batch_dataframes.append(batch_df)
            
            # 清理chunk列表
            batch_chunks = []
            gc.collect()
        else:
            print(f"  本批次无数据")
        
        print(f"  累计处理 {total_processed_stocks}/{len(stock_list)} 只股票")
        
        # 检查是否达到保存条件
        if batch_counter >= batches_per_save and batch_dataframes:
            print(f"\n✓ 达到保存条件，开始保存第 {file_index} 个文件...")
            print(f"  已处理 {batch_counter} 个批次，累积 {total_data_rows} 行数据")
            
            # 合并所有批次的DataFrame
            combined_df = pd.concat(batch_dataframes, ignore_index=True)
            print(f"  合并后数据: {len(combined_df)} 行")
            
            # 保存数据
            if save_to_excel_final(combined_df, file_index):
                file_index += 1
                
                # 清理内存：释放DataFrame和列表
                batch_dataframes = []
                batch_counter = 0
                total_data_rows = 0
                
                # 删除合并后的DataFrame
                del combined_df
                
                # 强制垃圾回收
                gc.collect()
                time.sleep(1)  # 暂停1秒，让系统有时间释放资源
                print("  ✓ 内存已清理")
            else:
                print("  ✗ 保存失败，保留当前数据继续处理")
        
        # 每处理完一批，进行基本的内存清理
        gc.collect()
        time.sleep(0.5)  # 暂停0.5秒，减少CPU占用
        
        # 每处理5批打印一次统计信息
        if current_batch % 5 == 0:
            print(f"\n--- 进度统计 ---")
            print(f"已处理: {total_processed_stocks}/{len(stock_list)} 只股票")
            print(f"无数据股票数: {len(no_data_stocks)}")
            print(f"出错股票数: {len(error_stocks)}")
            print(f"当前累积批次: {batch_counter}/{batches_per_save}")
            print(f"已生成文件: {file_index-1} 个")
            print("-" * 20)
    
    # 保存剩余数据
    if batch_dataframes:
        print(f"\n处理完所有股票，保存剩余数据...")
        combined_df = pd.concat(batch_dataframes, ignore_index=True)
        print(f"  剩余数据: {len(combined_df)} 行")
        
        if save_to_excel_final(combined_df, file_index):
            file_index += 1
            batch_dataframes = []
        
        del combined_df
    
    # 最终清理内存
    gc.collect()
    
    # 打印最终统计
    print(f"\n{'='*50}")
    print(f"处理完成！")
    print(f"总计处理股票: {len(stock_list)} 只")
    print(f"无数据股票: {len(no_data_stocks)} 只")
    if no_data_stocks and len(no_data_stocks) < 20:
        print(f"无数据股票列表: {', '.join(no_data_stocks[:20])}")
    print(f"出错股票: {len(error_stocks)} 只")
    if error_stocks:
        print(f"出错股票列表: {', '.join(error_stocks[:10])}")
    print(f"生成文件总数: {file_index-1} 个")
    print(f"数据保存在 'data/' 目录下")
    print(f"{'='*50}")

    print(f"[3] 开始获取十大流通股东数据...")
    # 处理股票数据
